    \subsection{Group Theory Basics}
    
    \begin{defi}{Group}
        A \textbf{group} $\group{G,*}$ is a set $G$, closed under a binary operaton $*$, which satisfies 
        \begin{itemize}
            \item associativity of $*$
            \item existence of identity element $e$
            \item existence of inverse element $a'$ of $a$ for all $a\in G$ 
        \end{itemize}
    \end{defi}
    \bigskip

    \begin{defi}{Homomorphism and Isomorphism}
        Let $\group{G_1,*_1}$ and $\group{G_2,*_2}$ be groups and $f:G_1\to G_2$. We say that $f$ is a \textbf{group isomorphism} if
        \begin{itemize}
            \item $f$ is bijective.
            \item For all $a,b\in G$, $f(a*_1b)=f(a)*_2f(b)$.
        \end{itemize}
        $f$ is a \textbf{group homomorphism} if the second condition is satisfied.
    \end{defi}
    \bigskip
    
    \begin{defi}{Permutation and Symmetric Group}
        Let $A \noteq \emptyset$. A \textbf{permutation} of $A$ is a bijection $\sigma \colon A \to A$. The \textbf{symmetric group} on $A$ is
        \[
            S_A = \operatorname{Sym}(A) = \set{\sigma \colon A \to A \mid \sigma \text{ is bijective}}
        \]
        For $A = \set{1, 2, \ldots, n}$, denote $S_n = S_{\{1,\ldots,n\}}$ with order $\abs{S_n} = n!$.
    \end{defi}
    \bigskip
    
    \begin{defi}{Cycle and Transposition}
        A \textbf{$k$-cycle} $(a_1 \; a_2 \; \cdots \; a_k) \in S_n$ ($k \ge 2$) is defined by
        \[
            \sigma(a_i) = a_{i+1} \ (1 \le i < k), \quad \sigma(a_k) = a_1, \quad \sigma(x) = x \ \forall x \notin \set{a_1, \ldots, a_k}.
        \]
        A $2$-cycle is a \textbf{transposition}. Inverses satisfy:
        \[
            (a_1 \; a_2 \; \cdots \; a_k)^{-1} = (a_k \; a_{k-1} \; \cdots \; a_1).
        \]
    \end{defi}
    \bigskip
    
    \begin{prop}
        Let $\supp(\sigma) = \set{x \in \set{1,\ldots,n} \mid \sigma(x) \noteq x}$.
        \begin{enumerate}
            \item \textbf{Disjoint Commutativity}: $\supp(\sigma) \cap \supp(\tau) = \emptyset \implies \sigma\tau = \tau\sigma$.
            \item \textbf{Disjoint Cycle Decomposition}: $\forall \pi \in S_n$, $\exists!$ partition of moved elements into disjoint cycles $c_1, \ldots, c_m$ such that
            \[
                \pi = c_1 c_2 \cdots c_m.
            \]
        \end{enumerate}
    \end{prop}
    \bigskip
    
    \begin{defi}{General Linear Group}
        For $n \in \Z^+$,
        \[
            \operatorname{GL}(n, \R) = \set{A \in M_n(\R) \mid \det(A) \noteq 0} = M_n(\R)^\times.
        \]
        $(\operatorname{GL}(n, \R), \cdot)$ is nonabelian for all $n \ge 2$.
    \end{defi}
    \bigskip
    
    \begin{defi}{Dihedral Group}
        For $n \ge 3$, the \textbf{dihedral group} $D_n$ is the isometry group of a regular $n$-gon in $\R^2$:
        \[
            D_n = \langle \rho, \mu \mid \rho^n = e, \; \mu^2 = e, \; \mu\rho\mu = \rho^{-1} \rangle.
        \]
    \end{defi}
    \bigskip
    
    \begin{thm}
        Let $n \ge 3$. Then $\abs{D_n} = 2n$, with canonical form:
        \[
            D_n = \set{\rho^k, \mu\rho^k \mid 0 \le k < n},
        \]
        governed by:
        \[
            \rho^k\mu = \mu\rho^{-k} = \mu\rho^{n-k}, \quad (\rho^k)^{-1} = \rho^{n-k}, \quad (\mu\rho^k)^{-1} = \mu\rho^k.
        \]
    \end{thm}
    \bigskip
    
    \begin{defi}{Subgroup}
        Let $(G, \cdot)$ be a group. $H \subseteq G$ is a \textbf{subgroup} ($H \le G$) if
        \[
            H \noteq \emptyset \aand (H, \cdot\res{H \times H}) \text{ is a group}.
        \]
        $H < G$ denotes a proper subgroup ($H \noteq G$). $H = \set{e}$ is the \textbf{trivial subgroup}; $H = G$ is the \textbf{improper subgroup}.
    \end{defi}
    
    \begin{thm}[Subgroup Criteria]
        Let $G$ be a group and $H \subseteq G$.
        \begin{enumerate}
            \item \textbf{Three-Step Test}:
            \[
                H \le G \iff (\forall a,b \in H \implies ab \in H) \land (e \in H) \land (\forall a \in H \implies a^{-1} \in H).
            \]
            \item \textbf{One-Step Test}:
            \[
                H \le G \iff H \noteq \emptyset \land (\forall a,b \in H \implies ab^{-1} \in H).
            \]
            \item \textbf{Finite Subgroup Test}: If $\abs{H} < \infty$ and $H \noteq \emptyset$,
            \[
                H \le G \iff (\forall a, b \in H \implies ab \in H).
            \]
        \end{enumerate}
    \end{thm}
    
    \begin{eg}[Standard Subgroups]
        \begin{enumerate}
            \item \textbf{Center}: $Z(G) = \set{z \in G \mid zg = gz \; \forall g \in G} \le G$ is abelian.
            \item \textbf{Special Linear Group}: $\operatorname{SL}(n, \R) = \set{A \mid \det(A) = 1} \le \operatorname{GL}(n, \R)$.
            \item \textbf{Roots of Unity}: $U_n = \set{z \in \C \mid z^n = 1} \le U = \set{z \in \C : \abs{z} = 1} \le \C^\times$.
        \end{enumerate}
    \end{eg}
    \bigskip
    
    \begin{defi}{Cyclic Subgroup and Generator}
        For $a \in G$, the \textbf{cyclic subgroup generated by $a$} is
        \[
            \langle a \rangle = \set{a^n \mid n \in \Z} \le G \quad (\text{in additive notation, } \set{na \mid n \in \Z}).
        \]
        $G$ is \textbf{cyclic} if $\exists a \in G$ such that $G = \langle a \rangle$; $a$ is called a \textbf{generator} of $G$.
    \end{defi}
    \bigskip
    
    \begin{thm}
        $G = \langle a \rangle \implies G \text{ is abelian}$.
    \end{thm}
    \bigskip
    
    \begin{thm}[Division Algorithm for $\Z$]
        \[
            \forall m \in \Z^+, \, \forall n \in \Z, \quad \exists! \, q, r \in \Z \text{ such that } n = mq + r \aand 0 \le r < m.
        \]
    \end{thm}
    \bigskip
    
    \begin{thm}[Subgroups of Cyclic Groups]
        Let $G = \langle a \rangle$.
        \begin{enumerate}
            \item $H \le G \implies H \text{ is cyclic}$.
            \item The subgroups of $(\Z, +)$ are precisely $n\Z = \langle n \rangle$ for $n \in \Z_{\ge 0}$.
        \end{enumerate}
    \end{thm}
    \bigskip
    
    \begin{defi}{Greatest Common Divisor}
        For $r \in \Z^+, s \in \Z_{\ge 0}$, $\gcd(r, s)$ is the unique positive generator of $r\Z + s\Z$:
        \[
            \langle \gcd(r, s) \rangle = \set{nr + ms \mid n, m \in \Z} \implies \exists n, m \in \Z \colon \gcd(r, s) = nr + ms.
        \]
    \end{defi}
    \bigskip
    
    \begin{thm}[Classification of Cyclic Groups]
        Let $G = \langle a \rangle$.
        \begin{enumerate}
            \item $\abs{G} = \infty \implies G \cong (\Z, +)$ via $a^k \mapsto k$.
            \item $\abs{G} = n < \infty \implies G \cong (\Z_n, +_n)$ via $a^k \mapsto k \pmod n$.
        \end{enumerate}
    \end{thm}
    \bigskip
    
    \begin{thm}[Orders, Generators, and Subgroup Lattice]
        Let $G = \langle a \rangle$ with $\abs{G} = n$, and let $b = a^s \in G$.
        \begin{enumerate}
            \item \textbf{Element Order}:
            \[
                \abs{a^s} = \abs{\langle a^s \rangle} = \frac{n}{\gcd(n, s)}, \aand \langle a^s \rangle = \langle a^{\gcd(n, s)} \rangle.
            \]
            \item \textbf{Generators}:
            \[
                \langle a^s \rangle = G \iff \gcd(n, s) = 1.
            \]
            The number of distinct generators of $G$ is $\phi(n) = \abs{\set{k \in \set{1,\ldots,n} \mid \gcd(n, k) = 1}}$.
            \item \textbf{Lagrange Divisibility}: $H \le G \implies \abs{H} \mid n$.
            \item \textbf{Subgroup Uniqueness}: For each divisor $d \mid n$ ($d > 0$), there exists a \textbf{unique} subgroup of order $d$:
            \[
                H_d = \langle a^{n/d} \rangle = \set{x \in G \mid x^d = e}, \quad \abs{H_d} = d.
            \]
        \end{enumerate}
    \end{thm}
    \bigskip
    
    \begin{thm}[Subgroup Intersections]
        Let $\set{H_i}_{i \in I}$ be any nonempty collection of subgroups of $G$. Then
        \[
            \bigcap_{i \in I} H_i \le G.
        \]
    \end{thm}
    \bigskip
    
    \begin{defi}{Subgroup Generated by a Subset}
        For $\emptyset \noteq S \subseteq G$:
        \[
            \group{S} = \bigcap \set{H \le G \mid S \subseteq H}.
        \]
        $S$ is a \textbf{generating set} of $G$ if $\langle S \rangle = G$. If $\exists S \subseteq G$ with $\abs{S} < \infty$ and $\langle S \rangle = G$, $G$ is \textbf{finitely generated}.
    \end{defi}
    \bigskip
    
    \begin{thm}[Constructive Characterization]
        For $\emptyset \noteq S \subseteq G$:
        \[
            \langle S \rangle = \set{x_1^{\epsilon_1} x_2^{\epsilon_2} \cdots x_k^{\epsilon_k} \mid k \in \N, \, x_j \in S, \, \epsilon_j \in \set{\pm 1}}.
        \]
    \end{thm}
    \bigskip

    \begin{defi}{Cayley Digraph}
        Let $G$ be a finite group and $S \subseteq G$ a generating set. The \textbf{Cayley digraph} $\Gamma(G, S) = (V, E)$ is defined by:
        \begin{itemize}
            \item $V = G$;
            \item $E = \set{(g, gs; s) \mid g \in G, \, s \in S}$, representing a directed arc $g \xrightarrow{s} gs$.
        \end{itemize}
        When $s = s^{-1}$, $(g, gs)$ and $(gs, g)$ collapse to an undirected edge.
    \end{defi}
    \bigskip
    
    \begin{prop}[Structural Properties of $\Gamma(G, S)$]
        \begin{enumerate}
            \item \textbf{Connectivity}: $\Gamma(G, S)$ is connected $\iff \langle S \rangle = G$.
            \item \textbf{Valency}: $\forall g \in G, \, \forall s \in S$, $\deg_{\text{in}}^{(s)}(g) = \deg_{\text{out}}^{(s)}(g) = 1$.
            \item \textbf{Vertex Transitivity}: $G \le \operatorname{Aut}(\Gamma(G, S))$ via left regular action $\lambda_h(g) = hg$.
            \item \textbf{Cycles and Relations}: Directed closed paths from $e$:
            \[
                e \xrightarrow{s_1^{\epsilon_1}} \cdots \xrightarrow{s_k^{\epsilon_k}} e \iff s_1^{\epsilon_1} \cdots s_k^{\epsilon_k} = e \text{ in } G.
            \]
        \end{enumerate}
    \end{prop}

    \begin{ex}
        Let $G$ be an arbitrary group and let $a,b\in G$.
        \begin{itemize}
            \item Prove that if $\abs{ab}=n<\infty$, then $\abs{ba}=n$.
            \begin{proof}
                Notice
                \[
                    (ab)^n=a(ba)^{n-1}b=e\implies a(ba)^{n}=ea=a\implies (ba)^{n}=e\implies \abs{ba}\leq n
                \]
                and
                \[
                    ((ba)^m=e\implies (ab)^m=e)\implies \abs{ba}\geq\abs{ab}=n
                \]
                Then $\abs{ba}=n$.
            \end{proof}
            \item Deduce that for any $g\in G$ and $x\in G$, $\abs{x}=\abs{gxg^{-1}}$
            \begin{proof}
                We have
                \[
                    (gxg^{-1})^m=gx^mg^{-1}=e\implies gx^m=g\implies x^m=e\implies \abs{gxg^{-1}}\geq\abs{x}
                \]
                We also have
                \[
                    x^m=e\implies gx^mg^{-1}=e\implies gxg^{-1})^m=e\implies \abs{x}\geq\abs{gxg^{-1}}
                \]
                Therefore, $\abs{x}=\abs{gxg^{-1}}$.
            \end{proof}
            \begin{remark}
                In fact, let $a=gx$ and $b=g^{-1}$ then $\abs{ab}=\abs{gxg^{-1}}=\abs{ba}=\abs{x}$.
            \end{remark}
        \end{itemize}
    \end{ex}

    \begin{ex}
        Let $H$ and $K$ be subgroups of a group $G$, and define the subset $HK=\set{hk:h\in H,k\in K}$
        \begin{itemize}
            \item Prove that if $G$ is abelian, then $HK\leq G$.
            \begin{proof}
                Let $hk,h'k'\in HK$, then we have
                \[
                    (hk)(h'k')^{-1}=hkk'^{-1}h'^{-1}=(hh'^{-1})(kk'^{-1})\in HK
                \]
                implying that $HK\leq G$.
            \end{proof}
            \item Give a counterexample when $G$ is nonabelian.
            \begin{proof}
                Let $G=D_3,H=\group{\mu},K=\group{\mu\rho}$, then $HK=\set{e,\mu,\rho,\mu\rho}$ is not a subgroup of $G$.
            \end{proof}
        \end{itemize}
    \end{ex}

    \begin{ex}
        Let $G$ be a finite cyclic group. Prove that if $H,K\leq G$ with $H\noteq K$, then $\abs{H}\noteq\abs{K}$.
    \end{ex}
    \begin{proof}
        Let $G=\group{a}$ and $\abs{G}=n$. Then $H,K\leq G$ implies $H=\group{a^p}=\group{a^{\gcd(n,p)}},K=\group{a^q}=\group{a^{\gcd(n,q)}}$ for some $p,q\in\Z^+$, then 
        \[
            H\neq K\implies\gcd(n,p)\neq\gcd(n,q)\implies\abs{H}=\frac{n}{\gcd(n,p)}\neq\frac{n}{\gcd(n,q)}=\abs{K}
        \]
    \end{proof}

    \begin{ex}
        Prove that a group $G$ with $\abs{G}>1$ has no proper nontrivial subgroups if and only if $G$ is finite and is isomorphic to $\Z_p$ for some prime $p$.
    \end{ex}
    \begin{proof}
        $\implies:$ Let $a\neq e\in G$.. Then $\group{a}\leq G$ must equals $G$ for otherwise it will be a proper nontrivial subgroup of $G$. Then $G$ is cyclic. If $G$ is infinite, then $\group{a}=\group{a^2}=G$ leads to a contradiction since $a\notin\group{a^2}$. If $\abs{G}=mn$ for some $m,n>1$, then $\abs{a^m}=mn/\gcd(mn,m)=n$ implying that $\group{a^m}$ is a proper nontrivial subgroup of $G$. Therefore, $G=\group{a}$ and $\abs{G}=p$ with $p$ prime. It follows that $G\cong\Z_p$.

        $\impliedby:$ Let $G\cong\Z_p$, then $G=\group{a}$ and $\abs{G}=p$. If $H\leq G$, then $H=\group{a^s}=\group{a^{\gcd(p,s)}}$. Since $p$ is prime, we know $H=\set{e}$ or $H=G$.
    \end{proof}
    
    \subsection{Finitely Generated Abelian Groups}
    
    \begin{defi}{Direct Product of Groups}
        Let $G_1, G_2, \ldots, G_n$ be groups. The \textbf{direct product} $\prod_{i=1}^n G_i = G_1 \times G_2 \times \cdots \times G_n$ is the Cartesian product endowed with componentwise binary operation:
        \[
            (a_1, a_2, \ldots, a_n)(b_1, b_2, \ldots, b_n) = (a_1 b_1, a_2 b_2, \ldots, a_n b_n).
        \]
        The identity element is $(e_1, e_2, \ldots, e_n)$, and $(a_1, \ldots, a_n)^{-1} = (a_1^{-1}, \ldots, a_n^{-1})$. It therefore forms a group.
    \end{defi}
    \bigskip

    \begin{thm}\label{0.2.2}
        The group $\prod_{i=1}^n \Z_{m_i}$ is cyclic and isomorphic to $\Z_{m_1 m_2 \cdots m_n}$ if and only if $\gcd(m_i, m_j) = 1$ for all $i \noteq j$.
    \end{thm}
    \bigskip

    \begin{thm}
        Let $(a_1, a_2, \ldots, a_n) \in \prod_{i=1}^n G_i$. If each $a_i$ has finite order $r_i$ in $G_i$, then
        \[
            \abs{(a_1, a_2, \ldots, a_n)} = \lcm(r_1, r_2, \ldots, r_n).
        \]
    \end{thm}
    \bigskip

    \begin{thm}[\textbf{Fundamental Theorem of Finitely Generated Abelian Groups - Primary Factor Form}]
        Every finitely generated abelian group $G$ is isomorphic to a direct product of cyclic groups:
        \[
            G \cong \Z_{(p_1)^{r_1}} \times \Z_{(p_2)^{r_2}} \times \cdots \times \Z_{(p_n)^{r_n}} \times \underbrace{\Z \times \cdots \times \Z}_{\beta \text{ copies}},
        \]
        where $p_i$ are primes (not necessarily distinct) and $r_i \in \Z^+$. $\beta$ is called the \textbf{Betti number} of $G$.
    \end{thm}
    \bigskip

    \begin{thm}[\textbf{Fundamental Theorem of Finitely Generated Abelian Groups - Invariant Factor Form}]
        Every finitely generated abelian group $G$ is isomorphic to a direct product of cyclic groups:
        \[
            G \cong \Z_{d_1} \times \Z_{d_2} \times \cdots \times \Z_{d_k} \times \underbrace{\Z \times \cdots \times \Z}_{\beta \text{ copies}},
        \]
        where each $d_i \ge 2$ and $d_1 \mid d_2 \mid \cdots \mid d_k$. The \textbf{invariant factors} $d_1, \ldots, d_k$ and the Betti number $\beta$ are uniquely determined.
    \end{thm}
    \bigskip

    \begin{defi}{Decomposable and Indecomposable Groups}
        A group $G$ is \textbf{decomposable} if it is isomorphic to the direct product of two proper nontrivial subgroups: $G \cong H \times K$ with $H, K \noteq \set{e}$. Otherwise, $G$ is \textbf{indecomposable}.
    \end{defi}
    \bigskip

    \begin{thm}
        \[
            G \text{ is finite indecomposable abelian} \iff G \cong \Z_{p^r} \text{ for some prime } p \text{ and } r \in \Z^+.
        \]
    \end{thm}
    \bigskip

    \begin{thm}
        If $G$ is a finite abelian group and $m \mid \abs{G}$, then $G$ contains a subgroup of order $m$.
    \end{thm}
    \bigskip

    \begin{thm}
        If $m \in \Z^+$ is square-free, then every abelian group of order $m$ is cyclic.
    \end{thm}
    \bigskip
    
    \begin{ex}
        Let $G=\Z_4\times\Z_{18}\times\Z_{15}$ and $H=\Z_3\times\Z_{36}\times\Z_{10}$. Are they isomorphic?
    \end{ex}
    \begin{proof}
        We have 
        \[
            G\cong\Z_4\times(\Z_2\times\Z_9)\times(\Z_3\times\Z_5)
        \]
        and 
        \[
            H\cong\Z_3\times(\Z_4\times\Z_9)\times(\Z_2\times\Z_5)
        \]
        It follows that $G\cong H$.
    \end{proof}
    \bigskip

    \begin{ex}
        Let $G$ be a finite abelian group. Prove that $G$ is not cyclic if and only if $G$ contains a subgroup isomorphic to $\Z_p\times\Z_p$ for some prime $p$.
    \end{ex}
    \begin{proof}
        Since $G$ is finite abelian, then
        \[
            G\cong\prod_{i=1}^n\Z_{(p_i)^{r_i}}
        \]
        By \hyperref[0.2.2]{Theorem 0.2.2}, if it is not cyclic, then $\gcd((p_i)^{r_i},(p_j)^{r_j})>1$ for some $i\neq j$, implying that $p_i=p_j$ and then
        \[
            \Z_{p_i}\times\Z_{p_i}\cong\group{(p_i)^{r_i-1}}\times\group{(p_j)^{r_j-1}}\leq\Z_{(p_i)^{r_i}}\times\Z_{(p_j)^{r_j}}\leq\prod_{i=1}^n\Z_{(p_i)^{r_i}}\cong G
        \]
        If $G$ is cylic, then $p_i\neq p_j$ for all $i\neq j$. It follows that if $H\leq G$, then
        \[
            H\cong\prod_{i=1}^nZ_{(p_i)^{q_i}},q_i\leq r_i
        \]
        which is not isomorphic to $\Z_p\times\Z_p$ since $p_i$ are mutually distinct.
    \end{proof}
    \begin{remark}
        In fact, when $G$ is cylic, we can deduce that $H\leq G$ is also cyclic, which directly tells proves the result.
    \end{remark}
    \begin{remark}
        Generally, one can prove that:
        \begin{quote}
            Let $G$ be a finite abelian group, and let $x_1,\dots,x_n\geq 2$ be integers. The group $H=\prod_{i=1}^n\Z_{x_i}$ is isomorphic to a subgroup of $G$ if and only if $G$ can be decomposed as a direct product of cyclic groups $G\cong\prod_{j=1}^m\Z_{y_j}$ such that $m\geq n$ and $x_i\mid y_i$ for all $i\in\set{1,\dots,n}$.
        \end{quote}
    \end{remark}

    \begin{ex}
        Let $G,H,K$ be finitely generated abelian groups. Show that
        \[
            G\times K\cong H\times K\implies G\cong H
        \]
    \end{ex}
    \begin{proof}
        Since $G,H,K$ are finitely generated abelian groups, so are $G\times K$ and $H\times K$. Since $G\times K\cong H\times K$, so they have the same Betti number and the same collection of $(p_i)^{r_i}$ after decomposition. It follows that after picking up the part given by $K$, we know $G$ and $H$ have the same Betti number and the same collection of $(p_i)^{r_i}$ after decomposition, which implies $G\cong H$.
    \end{proof}

    \subsection{Cosets and the Theorem of Lagrange}

    \begin{thm}
        Let $H \le G$. The relation $\sim_L$ defined on $G$ by
        \[
            a \sim_L b \iff a^{-1}b \in H
        \]
        is an equivalence relation on $G$. Similarly, $a \sim_R b \iff ab^{-1} \in H$ is an equivalence relation.
    \end{thm}
    \bigskip

    \begin{defi}{Coset}
        Let $H \le G$ and $a \in G$.
        \begin{itemize}
            \item The \textbf{left coset} of $H$ containing $a$ is the equivalence class under $\sim_L$:
            \[
                aH = \set{ah \mid h \in H}.
            \]
            \item The \textbf{right coset} of $H$ containing $a$ is the equivalence class under $\sim_R$:
            \[
                Ha = \set{ha \mid h \in H}.
            \]
        \end{itemize}
    \end{defi}
    \bigskip

    \begin{thm}
        Let $H \le G$. For every $a \in G$, the map $f \colon H \to aH$ defined by $f(h) = ah$ is a bijection. Consequently,
        \[
            \abs{aH} = \abs{H} = \abs{Ha} \quad (\forall a \in G).
        \]
    \end{thm}
    \bigskip

    \begin{thm}[Theorem of Lagrange]
        Let $H$ be a subgroup of a finite group $G$. Then the order of $H$ divides the order of $G$:
        \[
            \abs{H} \mid \abs{G}.
        \]
    \end{thm}
    
    \begin{cor}\label{0.3.5}
        Every group of prime order $p$ is cyclic, hence isomorphic to $\Z_p$.
    \end{cor}
    \bigskip

    \begin{thm}
        Let $G$ be a finite group. For every $g \in G$, $\abs{g}$ divides $\abs{G}$. In particular, $g^{\abs{G}} = e$.
    \end{thm}
    \bigskip

    \begin{defi}{Index of a Subgroup}
        Let $H \le G$. The \textbf{index} of $H$ in $G$, denoted $(G : H)$ or $[G : H]$, is the cardinality of the set of left (or right) cosets of $H$ in $G$. When $G$ is finite,
        \[
            [G : H] = \frac{\abs{G}}{\abs{H}}.
        \]
    \end{defi}
    \bigskip

    \begin{thm}[Tower Law for Subgroup Indices]
        Let $K \le H \le G$. If $[G : H]$ and $[H : K]$ are finite, then $[G : K]$ is finite and
        \[
            [G : K] = [G : H] \cdot [H : K].
        \]
    \end{thm}
    \bigskip

    \begin{thm}[Cosets of Homomorphism Kernels]
        Let $\phi \colon G \to G'$ be a group homomorphism. Then
        \begin{enumerate}
            \item Left and right cosets of $\ker(\phi)$ coincide:
            \[
                a \ker(\phi) = \ker(\phi) a \quad (\forall a \in G).
            \]
            \item Two elements $a, b \in G$ belong to the same coset of $\ker(\phi)$ if and only if $\phi(a) = \phi(b)$:
            \[
                a \ker(\phi) = b \ker(\phi) \iff \phi(a) = \phi(b).
            \]
        \end{enumerate}
    \end{thm}
    \bigskip

    \begin{cor}
        A group homomorphism $\phi \colon G \to G'$ is injective if and only if
        \[
            \ker(\phi) = \set{e_G}.
        \]
    \end{cor}
    \bigskip

    \begin{ex}
        Let $\sigma=(1,2,5,4)(2,3)$ in $S_5$, find the index of $\group{\sigma}$ in $S_5$.
    \end{ex}
    \begin{proof}
        Notice $\sigma=(1,2,3,5,4)$, then the order of $\sigma$ is $5$ in $S_5$, implying the index given by 
        \[
            [S_5:\group{\sigma}]=\frac{\abs{S_5}}{\abs{\group{\sigma}}}=\frac{5!}{5}=24
        \]
    \end{proof}
    \bigskip

    \begin{ex}
        Let $H$ be a subgroup of a group $G$, let $a,b\in G$. Is it true that 
        \[
            aH=bH\implies Ha^{-1}=Hb^{-1}
        \]
    \end{ex}
    \begin{proof}
        The statement is true. Since $aH=bH$, then $ah_a=bh_b$ for some $h_a,h_b\in H$, it follows that $b^{-1}a\in H$. Therefore, for $hb^{-1}\in Hb^{-1}$, we have $hb^{-1}=hb^{-1}aa^{-1}=(hb^{-1}a)a^{-1}\in Ha^{-1}$. Then $Hb^{-1}\subset Ha^{-1}$, similarly we can deduce that $H^{-1}\subset Hb^{-1}$, thus $Ha^{-1}=Hb^{-1}.$
    \end{proof}
    \bigskip

    \begin{ex}
        Let $G$ be a group of order $pq$, where $p$ and $q$ are prime numbers. Show that every proper subgroup of $G$ is cyclic.
    \end{ex}
    \begin{proof}
        By Theorem of Lagrange we know every proper subgroup $H$ has order $\abs{H}=n\mid pq$ implying $n=1$ or $n=p$ or $n=q$. If $n=1$ then $H$ is trivially cyclic, if $n=p$ or $n=q$ then \hyperref[0.3.5]{Corollary 0.3.5} tells that $H$ is cyclic.         
    \end{proof}

    \subsection{Normal Subgroups and Factor Groups}

    \begin{defi}{Normal Subgroup}
        A subgroup $N \le G$ is \textbf{normal} in $G$, written
        $N \normal G$, if
        \[
            gN = Ng \qquad (\forall g \in G)
        \]
    \end{defi}
    \bigskip
    
    \begin{thm}[Characterizations of Normality]
        Let $N \le G$. The following are equivalent:
        \begin{enumerate}
            \item $N \normal G$.
            \item $gNg^{-1} = N$ for every $g \in G$.
            \item $gng^{-1} \in N$ for every $g \in G$ and $n \in N$.
            \item There is a group homomorphism $\phi:G\to G'$ such that $\ker(\phi)=N$.
        \end{enumerate}
    \end{thm}
    
    \begin{cor}
        Every subgroup of an abelian group is normal.
    \end{cor}
    \bigskip
    
    \begin{defi}{Factor Group}
        Let $N \normal G$. The \textbf{factor group}, or
        \textbf{quotient group}, of $G$ by $N$ is
        \[
            G/N = \set{gN \mid g \in G},
            \qquad
            (aN)(bN) = (ab)N.
        \]
        Its identity element is $N$, and
        \[
            (gN)^{-1} = g^{-1}N.
        \]
    \end{defi}
    \bigskip
    
    \begin{defi}{Canonical Projection}
        For $N \normal G$, the map
        \[
            q \colon G \to G/N,
            \qquad
            q(g) = gN,
        \]
        is called the \textbf{canonical projection}.
    \end{defi}
    \bigskip
    
    \begin{thm}
        The canonical projection $q \colon G \to G/N$ is a
        surjective homomorphism with
        \[
            \ker(q) = N.
        \]
    \end{thm}
    \bigskip
    
    \begin{thm}[First Isomorphism Theorem]
        Let $\phi \colon G \to G'$ be a group homomorphism, and set
        $K = \ker(\phi)$. Then
        \[
            \bar{\phi} \colon G/K \to \operatorname{im}(\phi),
            \qquad
            \bar{\phi}(gK) = \phi(g),
        \]
        is an isomorphism. Thus
        \[
            \boxed{G/\ker(\phi) \cong \image(\phi).}
        \]
        If $q \colon G \to G/K$ is the canonical projection, then
        $\bar{\phi}$ is the unique map satisfying
        \[
            \bar{\phi}(q(g)) = \phi(g)
            \qquad (\forall g \in G).
        \]
        In particular, if $\phi$ is surjective, then
        \[
            G/\ker(\phi) \cong G'.
        \]
    \end{thm}
    \bigskip
    
    \begin{remark}
        To identify a quotient $G/N$, it is often useful to construct a
        surjective homomorphism $\phi \colon G \to H$ with
        $\ker(\phi)=N$. The First Isomorphism Theorem then gives
        $G/N \cong H$.
    \end{remark}
    \bigskip
    
    \begin{defi}{Automorphism and Inner Automorphism}
        An \textbf{automorphism} of $G$ is an isomorphism $G \to G$.
        For each $g \in G$, the map
        \[
            i_g \colon G \to G,
            \qquad
            i_g(x) = gxg^{-1},
        \]
        is an automorphism, called the \textbf{inner automorphism}·
        induced by $g$. The operation $x \mapsto gxg^{-1}$ is called
        \textbf{conjugation} by $g$.
    \end{defi}
    \bigskip
    
    \begin{prop}
        A subgroup $N \le G$ is normal if and only if it is invariant
        under every inner automorphism:
        \[
            i_g(N) = N \qquad (\forall g \in G).
        \]
    \end{prop}
    \bigskip
    
    \begin{remark}
        Normality means that conjugation preserves $N$ as a set.
        It does not require $gng^{-1}=n$ for every $g \in G$ and
        $n \in N$.
    \end{remark}
    \bigskip

    \begin{ex}
        Let $F$ be the group of all functions $\R\to\R$, under pointwise addition:
        \[
            (f+g)(x)=f(x)+g(x)
        \]
        Fix $c\in\R$, and define
        \[
            \phi_c:F\to\R,\qquad \phi_c(f)=f(c).
        \]
        \begin{itemize}
            \item Prove that $\phi_c$ is a group homomorphism.
            \begin{proof}
                $\phi_c$ is a homomorphism since for $f,g\in F$, we have
                \[
                    \phi_c(f+g)=(f+g)(c)=f(c)+g(c)=\phi_c(f)+\phi_c(g)
                \]
            \end{proof}
            \item Determine $\ker(\phi_c)$.
            \begin{proof}
                \[
                    \ker(\phi_c)=\set{f\in F\mid\phi_c(f)=0}=\set{f:\R\to\R\mid f(c)=0}
                \]
            \end{proof}
            \item Describe the coset of $\ker(\phi_c)$ containing the constant function $f(x)=1$.
            \begin{proof}
                The coset is given by 
                \[
                    f+\ker(\phi_c)=1+\set{f:\R\to\R\mid f(c)=0}=\set{f:\R\to\R\mid f(c)=1}
                \]
            \end{proof}
            \item Identify a familiar group isomorphic to $F/\ker(\phi_c)$.
            \begin{proof}
                By First Isomorphism Theorem, since $\phi_c:F\to\R$ is a group homomorphism, then
                \[
                    F/\ker(\phi_c)\cong\image(\phi_c)=\R
                \]
            \end{proof}
        \end{itemize}
    \end{ex}
    \bigskip

    \begin{ex}
        Let $H,N\leq G$ with $N\normal G$. Prove that $H\cap N\normal H$ and give a example showing that $H\cap N$ need not be normal in $G$.
    \end{ex}
    \begin{proof}
        Let $x\in H,y\in H\cap N$, then $xyx^{-1}\in H\cap N$ since
        \[
            x\in G,y\in N\implies xyx^{-1}\in N
        \]
        \[
            x,y\in H\implies xyx^{-1}\in H
        \]

        Let $G=D_3,N=G,H=\group{\mu}$, then $N\cap H=\group{\mu}$ is not a normal subgroup of $G$ since $\mu\rho\neq\rho\mu$.
    \end{proof}
    \bigskip

    \begin{ex}
        Let $G,G'$ be groups with $H\normal G$ and $H'\normal G'$. Let $\phi:G\to G'$ be a homomorphism satisfying:
        \[
            \phi(H)\subset H'
        \]
        Prove that $\phi$ induces a well-defined homomorphism
        \[
            \phi_*:G/H\to G'/H'\qquad\phi_*(gH)=\phi(g)H'
        \]
    \end{ex}
    \begin{proof}
        If $xH=yH$, then
        \[
            xy^{-1}\in H\implies \phi(xy^{-1})\in H'\implies \phi(x)H'=\phi(y)H'\implies \phi_*(xH)=\phi_*(yH)
        \]
        which implies $\phi_*$ is well-defined. Since $\phi$ is a homomorphism, then
        \[
            \phi_*(xyH)=\phi(xy)H'=(\phi(x)H')(\phi(y)H')=\phi_*(xH)\phi_*(yH)
        \]
        which implies $\phi_*$ is a homomorphism
    \end{proof}
    
    
    \subsection{Factor-Group Computations and Simple Groups}
    
    \begin{thm}[Quotients of Direct Products]
        Let $H$ and $K$ be groups. Then
        \[
            H \times \set{e_K} \normal H \times K,
            \qquad
            \set{e_H} \times K \normal H \times K,
        \]
        and the coordinate projections induce isomorphisms
        \[
            \frac{H \times K}{H \times \set{e_K}} \cong K,
            \qquad
            \frac{H \times K}{\set{e_H} \times K} \cong H.
        \]
    \end{thm}
    \bigskip
    
    \begin{thm}[Cyclic and Abelian Quotients]
        Let $N \normal G$.
        \begin{enumerate}
            \item If $G$ is cyclic, then $G/N$ is cyclic. More precisely,
            \[
                G = \group{a}
                \implies G/N = \group{aN}.
            \]
            \item If $G$ is abelian, then $G/N$ is abelian.
        \end{enumerate}
    \end{thm}
    \bigskip
    
    \begin{eg}
        For every group $G$ and every positive integer $n$,
        \[
            G/\set{e_G} \cong G,
            \qquad
            G/G \cong \set{e},
            \qquad
            \Z/n\Z \cong \Z_n.
        \]
    \end{eg}
    \bigskip
    
    \begin{thm}[Images and Preimages of Normal Subgroups]
        Let $\phi \colon G \to G'$ be a group homomorphism.
        \begin{enumerate}
            \item If $N \normal G$, then
            \[
                \phi(N) \normal \phi(G).
            \]
            \item If $M \normal \phi(G)$, then
            \[
                \phi^{-1}(M) \normal G.
            \]
            \item In particular, if $M \normal G'$, then
            $\phi^{-1}(M) \normal G$.
        \end{enumerate}
    \end{thm}
    \bigskip
    
    \begin{remark}
        The image of a normal subgroup is normal in $\phi(G)$,
        but need not be normal in the whole codomain $G'$.
    \end{remark}
    \bigskip
    
    \begin{defi}{Simple Group}
        A group $G$ is \textbf{simple} if $G \neq \set{e}$ and its
        only normal subgroups are $\set{e}$ and $G$.
    \end{defi}
    \bigskip
    
    \begin{thm}
        An abelian group is simple if and only if it is isomorphic
        to $\Z_p$ for some prime $p$.
    \end{thm}
    \bigskip
    
    \begin{thm}[Simplicity of Alternating Groups]
        The alternating group $A_n$ is simple for every $n \ge 5$.
    \end{thm}
    \bigskip
    
    \begin{defi}{Maximal Normal Subgroup}
        A \textbf{maximal normal subgroup} of $G$ is a proper normal
        subgroup $M \normal G$ such that there is no normal subgroup
        $N \normal G$ satisfying
        \[
            M < N < G.
        \]
    \end{defi}
    \bigskip
    
    \begin{thm}
        Let $M \normal G$. Then
        \[
            \boxed{
                M \text{ is maximal normal in } G
                \iff G/M \text{ is simple}.
            }
        \]
    \end{thm}
    \bigskip
    
    \begin{remark}
        A maximal normal subgroup is maximal among proper
        \emph{normal} subgroups. It need not be maximal among all
        proper subgroups.
    \end{remark}
    \bigskip
    
    \begin{defi}{Center of a Group}
        The \textbf{center} of $G$ is
        \[
            Z(G) = \set{z \in G \mid zg=gz
            \text{ for every } g \in G}.
        \]
    \end{defi}
    \bigskip
    
    \begin{prop}
        The center $Z(G)$ is an abelian normal subgroup of $G$.
        Moreover,
        \[
            G \text{ is abelian} \iff Z(G)=G.
        \]
    \end{prop}
    \bigskip
    
    \begin{defi}{Commutator and Commutator Subgroup}
        For $a,b \in G$, the element
        \[
            [a,b] = aba^{-1}b^{-1}
        \]
        is called their \textbf{commutator}.
        The \textbf{commutator subgroup} of $G$ is
        \[
            [G,G] = \group{[a,b] \mid a,b \in G}.
        \]
    \end{defi}
    \bigskip
    
    \begin{remark}
        The commutator subgroup is generated by all commutators.
        The set of individual commutators need not itself be a subgroup.
    \end{remark}
    \bigskip
    
    \begin{thm}
        The commutator subgroup satisfies $[G,G] \normal G$.
        For every $N \normal G$,
        \[
            \boxed{
                G/N \text{ is abelian} \iff [G,G] \le N.
            }
        \]
        Thus $[G,G]$ is the smallest normal subgroup of $G$
        whose quotient is abelian.
    \end{thm}
    \bigskip
    
    \begin{defi}{Abelianization}
        The \textbf{abelianization} of $G$ is the abelian group
        \[
            G_{\mathrm{ab}} = G/[G,G].
        \]
    \end{defi}
    \bigskip
    
    \begin{cor}[Universal Property of Abelianization]
        Let $A$ be an abelian group and let $\phi \colon G \to A$
        be a homomorphism. There is a unique homomorphism
        \[
            \bar{\phi} \colon G_{\mathrm{ab}} \to A
        \]
        such that $\phi=\bar{\phi}\circ q$, where
        $q \colon G \to G_{\mathrm{ab}}$ is the canonical projection.
        Explicitly,
        \[
            \bar{\phi}(g[G,G])=\phi(g).
        \]
    \end{cor}
    \bigskip

    \begin{ex}
        Classify the group
        \[
            \frac{\Z\times\Z\times\Z}{\group{(3,3,3)}}
        \]
        according to the Fundamental Theorem of Finitely Generated Abelian Groups.
    \end{ex}
    \begin{proof}
        Define $\phi:\Z\times\Z\times\Z\to\Z_3\times\Z\times\Z$ by $\phi(x,y,z)=((x+y-z)\mod 3,x-y,y-z)$ be a homormophism with $\ker(\phi)=\group{(3,3,3)}$. It follows that 
        \[
            \frac{\Z\times\Z\times\Z}{\group{(3,3,3)}}\cong\image(\phi)=\Z_3\times\Z\times\Z
        \]
    \end{proof}

    \begin{ex}
        Find both the center $Z(D_4)$ and the commutator subgroup $[D_4,D_4]$ of the group $D_4$.
    \end{ex}
    \begin{proof}
        One can varify that $Z(D_4)=\set{e,\rho^2}$.
        
        Let $N=\group{\rho^2}\normal D_4$. Notice for $a^2\in N$ for all $a\in D_4$, implying that $aN=a^{-1}N$. It follows that for $aN,bN\in D_4/N$,

            abN=a^{-1}b^{-1}N=(ba)^{-1}N=baN

            G/Z(G)

    \subsection{Isomorphism Theorems}
    
    % The First Isomorphism Theorem is stated above.
    
    \begin{lemma}[Normal-Subgroup Correspondence]
        Let $N \normal G$, and let $q \colon G \to G/N$ be the
        canonical projection. There is an inclusion-preserving bijection
        \[
            \set{L \normal G \mid N \le L}
            \longleftrightarrow
            \set{M \normal G/N}
        \]
        given by
        \[
            L \longmapsto L/N,
            \qquad
            M \longmapsto q^{-1}(M).
        \]
        These assignments are mutually inverse.
    \end{lemma}
    \bigskip
    
    \begin{defi}{Product and Join of Subgroups}
        Let $H,N \le G$. Their \textbf{set product} is
        \[
            HN = \set{hn \mid h \in H,\ n \in N}.
        \]
        Their \textbf{join} is
        \[
            H \vee N = \group{H \cup N},
        \]
        the smallest subgroup of $G$ containing both $H$ and $N$.
    \end{defi}
    \bigskip
    
    \begin{lemma}
        Let $H \le G$ and $N \normal G$. Then
        \[
            H \vee N = HN = NH \le G.
        \]
        If $H$ is also normal in $G$, then $HN \normal G$.
    \end{lemma}
    \bigskip
    
    \begin{remark}
        Without additional hypotheses, the set product of two
        subgroups need not be a subgroup.
    \end{remark}
    \bigskip
    
    \begin{thm}[Second Isomorphism Theorem]
        Let $H \le G$ and $N \normal G$. Then
        \[
            H \cap N \normal H,
            \qquad
            N \normal HN,
        \]
        and there is an isomorphism
        \[
            \boxed{H/(H \cap N) \cong HN/N}
        \]
        given by
        \[
            h(H \cap N) \longmapsto hN.
        \]
    \end{thm}
    \bigskip
    
    \begin{remark}
        The homomorphism
        \[
            H \to HN/N,\qquad h \mapsto hN,
        \]
        is surjective and has kernel $H \cap N$.
        Thus the Second Isomorphism Theorem identifies the image
        of $H$ under the quotient map $G \to G/N$.
    \end{remark}
    \bigskip
    
    \begin{thm}[Third Isomorphism Theorem]
        Let $K,H \normal G$ with $K \le H$. Then
        \[
            H/K \normal G/K
        \]
        and there is an isomorphism
        \[
            \boxed{(G/K)/(H/K) \cong G/H}
        \]
        given by
        \[
            (gK)(H/K) \longmapsto gH.
        \]
    \end{thm}
    \bigskip
    
    \begin{remark}
        The homomorphism
        \[
            G/K \to G/H,\qquad gK \mapsto gH,
        \]
        is surjective and has kernel $H/K$.
        Quotienting first by $K$ and then by $H/K$ therefore has
        the same effect as quotienting directly by $H$.
    \end{remark}
    \bigskip

    \begin{ex}
        Let $H,K$ and $L$ be normal subgroups of $G$ with $H<K<L$. Let $A=G/H,B=K/H,C=L/H$
        \begin{itemize}
            \item Show that $B$ and $C$ are normal subgroups of $A$, and $B<C$.
            \begin{proof}
                For $gH\in A,kH\in B$, we have $gHkHg^{-1}H=(gkg^{-1})H=k'H$ for some $k'\in K$, hence $B\normal A$. Similarly, we know $C\normal A$.
                
                For $kH\in B$, we know $k\in L$ thus $kH\in C$; $K<L$ means there is $l\in L\setminus K$ thus $lH\in C\setminus B$. It follows that $B<C$.
            \end{proof}
            \item To what factor group of $G$ is $(A/B)/(C/B)$ isomorphic?
            \begin{proof}
                By Third Isomorphism Theorem we know
                \[
                    (A/B)/(C/B)\cong A/C=(G/H)/(L/H)\cong G/L  
                \]
            \end{proof}
        \end{itemize}
    \end{ex}
    \bigskip

    \begin{ex}
        Let $K$ and $L$ be normal subgroups of $G$ with $K\vee L=G$, and $K\cap L=\set{e}$. Show that $G/K\cong L$ and $G/L\cong K$.
    \end{ex}
    \begin{proof}
        Since $K\normal G$ and $L\normal G$, we know $G=K\vee L=KL$. Then the Second Isomorphism Theorem implies that
        \[
            KL/K=G/K\cong L/(K\cap L)=L/\set{e}\cong L
        \]
        Similar argument shows that $G/L\cong K$.
    \end{proof}
    \bigskip

    \begin{ex}
        Let $G$ be a group with subgroups $H^*\leq H\leq G$ and $K^*\leq K\leq G$. Prove that
        \[
            H^*(H\cap K^*)\cap(H\cap K)=(H^*\cap K)(H\cap K^*)
        \]
    \end{ex}
    \begin{proof}
        Notice
        \[
            H^*(H\cap K^*)\cap(H\cap K)=((H^*\cap K)\cup(H^*\setminus K))(H\cap K^*)\cap(H\cap K)
        \]
        Since $H\cap K\leq G$, then for $x\in H\cap K^*,y\in H\cap K$, one has $yx^{-1}\in H\cap K$, it follows that
        \[
            (H^*\setminus K)(H\cap K^*)\cap(H\cap K)=\emptyset
        \]
        then
        \[
            H^*(H\cap K^*)\cap(H\cap K)=(H^*\cap K)(H\cap K^*)\cap(H\cap K)=(H^*\cap K)(H\cap K^*)
        \]
        The second equation is due to $H^*\cap K\leq H\cap K$ and $H\cap K^*\leq H\cap K$.
    \end{proof}

    \subsection{Homotopy of Paths}
    \begin{defi}{Homotopic and Nulhomotopic}
        If $f$ and $f'$ are continuous map of the space $X$ into the space $Y$, we say $f$ is \textbf{homotopic} to $f'$ if there is a continuous map $F:X\times I\to Y$ such that 
        \[
            F(x,0)=f(x)\aand F(x,1)=f'(x)
        \]
        for each $x$. The map $F$ is called a \textbf{homotopy} between $f$ and $f'$, denoted by $f\simeq f'$. If $f\simeq f'$ and $f'$ is a constant map, we say that $f$ is \textbf{nulhomotopic}.
    \end{defi}
    
    \begin{defi}{Path Homotopic}
        Two paths $f,f':I\to X$ are said to be \textbf{path homotopic} if they have the same initial point $x_0$ and the same final point $x_1$, and if there is a continuous map $F:I\times I\to X$ such that
        \[
            F(s,0)=f(s)\aand F(s,1)=f'(s)
        \]
        \[
            F(0,t)=x_0\aand F(1,t)=x_1
        \]
        for each $s,t\in I$. The map $F$ is called a \textbf{path homotopy} between $f$ and $f'$, denoted by $f\simeq_p f'$.
    \end{defi}
    \begin{remark}
        In this case, the relations $\simeq$ and $\simeq_p$ are equivalence relations.

        In particular, if $f$ is a path, then its path homotopy equivalence class is denoted by $[f]$.
    \end{remark}
    \begin{remark}
        Note that generally it's not true that for paths $f,g$ which share the same endpoints, they are homotopic iff they are path homotopic.
        
        The reason is path homotopy requires endpoints to be fixed during the transformation while homotopy does not. As a result, if there is a hole between two endpoints, then path homotopy cannot transform a path on the one side to the other side while homotopy can.

        In fact, a path $f$ is always homotopic to the constant map at $f(0)$, thus being nulhomotopic. As a result, we have:
        \begin{quote}
            Paths $f,g:I\to X$ are homotopic if and only if $f(0)$ and $g(0)$ belong to the same path component of $X$.
        \end{quote}
    \end{remark}
    \begin{remark}\addelement{Straight-Line Homotopy}
        However, let $A$ be any convex subspace of $\R^n$. Then any two paths $f,g:I\to A$ which share the same endpoint are path homomoprhic under the \textbf{straight-line homotopy}:
        \[
            F(x,t)=(1-t)f(x)+tg(x)
        \]
    \end{remark}
    \bigskip

    \begin{defi}{Product of Paths}
        Let $f$ be a path in $X$ from $x_0$ to $x_1$, $g$ be a path in $X$ from $x_1$ to $x_2$. We define the \textbf{product} $f*g$ of $f$ and $g$ to be the path $h$ from $x_0$ to $x_2$ given by
        \[
            h(s)=\begin{cases}
                f(2s)&s\in[0,\frac{1}{2}]\\
                g(2s-1)&s\in[\frac{1}{2},1]
            \end{cases}
        \]
        It then induces a well-defined operation $*$ on path-homotopy classes defined by
        \[
            \boxed{[f]*[g]=[f*g]}
        \]
    \end{defi}
    \begin{thm}[Groupoid Properties]
        The operation $*$ on path-homotopy classes has the following properties:
        \begin{itemize}
            \item (Associativity) With endpoints match, one has
            \[
                ([f]*[g])*[h]=[f]*([g]*[h])
            \]
            \item (Right and left identities) Let $e_x(s)=x$ be the constant path and $f$ be a path from $x_0$ to $x_1$, then
            \[
                [f]*[e_{x_1}]=[f]=[e_{x_0}]*[f]
            \]
            \item Let $f$ be a path from $x_0$ to $x_1$, $\bar{f}(s)=f(1-s)$ be a path from $x_1$ to $x_0$, then
            \[
                [f]*[\bar{f}]=[e_{x_0}]\aand [\bar{f}]*[f]=[e_{x_1}]
            \]
        \end{itemize}
    \end{thm}
    \begin{remark}
        Let $f,f':I\to X$ be paths and $k:X\to Y$ be continuous. Then
        \[
            f\simeq_p f'\implies k\circ f\simeq_p k\circ f'
        \]
    \end{remark}
    \begin{remark}
        Let $f,g:I\to X$ with $f(1)=g(0)$ and $k:X\to Y$ be continuous. Then
        \[
            k\circ(f*g)=(k\circ f)*(k\circ g)
        \]
    \end{remark}
    \bigskip

    \begin{thm}
        Let $f:I\to X$ be a path and
        \[
            0=a_0<a_1<\cdots<a_n=1
        \]
        Reparametrize each segment as a path on $I$:
        \[
            f_i(s)=f(a_{i-1}+(a_i-a_{i-1})s)
        \]
        Then
        \[
            [f]=[f_1]*\cdots*[f_n]
        \]
    \end{thm}
    \bigskip

    \begin{ex}
        Let $I=[0,1]$, then for any space $X$ and continuous maps $f,g:X\to I$. We have $f\simeq g$.
    \end{ex}
    \begin{proof}
        Define $F:X\times I\to I$ by the straight-line homotopy
        \[
            F(x,t)=(1-t)f(x)+tg(x)
        \]
        Then one can verify that $F$ is a homotopy between $f$ and $g$.
    \end{proof}
    \bigskip

    \begin{ex}\addelement{Contractible}
        A space $X$ is said to be \textbf{contractible} if the identity map $i_x:X\to X$ is nulhomotopic.
        \begin{itemize}
            \item Show that $I$ and $\R$ are contractible.
            \begin{proof}
                Since both $I$ and $\R$ are both convex subspaces of $\R$. Then $f(x)=0$ is homotopic to $i_x$ given by the striaght-line homotopy
                \[
                    F(x,t)=(1-t)f(x)+ti_x(x)
                \]
            \end{proof}
            \item Show that a contractible space is path connected.
            \begin{proof}
                Let $X$ be a contractible space and $i_x\simeq f$ where $f(x)=y$ for all $x\in X$. Let $F$ be the homotopy from $i_x$ to $f$. Then given any $z\in X$, the map $g:I\to X$ defined by
                \[
                    g(t)=F(z,t)
                \]
                is a path from $z$ to $y$, which implies $X$ is path connected.
            \end{proof}
            \item Let $f,g:X\to Y$ be continuous where $Y$ is contractible, show that $f\simeq g$.
            \begin{proof}
                Since $Y$ is contractible. Then $i_y\simeq h$ for some $h(y)=y_0$ with homotopy $F$ satisfying $F(y,0)=y$ and $F(y,1)=y_0$. Then $G:X\times I\to Y$ defined by
                \[
                    G(x,t)=F(f(x),t)
                \]
                is a homotopy from $f$ to $u(x)=y_0$. It follows that $f\simeq g$.
            \end{proof}
            \item Let $f,g:X\to Y$ be continuous where $X$ is contractible and $Y$ is path connected, show that $f\simeq g$.
            \begin{proof}
                Let $i_x\simeq u$ with $u(x)=x_0$ and homotopy $U$ such that $U(x,0)=x$ and $U(x,1)=x_0$. Define $h_f:X\to Y$ by $h_f(x)=f(x_0)$. We know $f\simeq h_f$ under the homotopy defined by
                \[
                    F(x,t)=f(U(x,t))
                \]
                Similarly, $g\simeq h_g$. Finally, there is a path $p:I\to Y$ from $f(x_0)$ to $g(x_0)$, which induces the homotopy 
                \[
                    P(x,t)=p(t)
                \]
                implying that $h_f\simeq h_g$. Therefore $f\simeq g$.
            \end{proof}
        \end{itemize}
    \end{ex}

    \subsection{The Fundamental Group}
    \begin{defi}{Fundamental Group}
        Let $X$ be a space; let $x_0\in X$. A path in $X$ that begins and ends at $x_0$ is called a \textbf{loop} based on $x_0$.

        The set of path homotopy classes of loops based at $x_0$, with the operation $*$, form a group, called the \textbf{fundamental group} of $X$ relative to the base point $x_0$, denoted by $\pi_1(X,x_0)$.
    \end{defi}
    \begin{remark}
        Sometimes this group is called the \textbf{first homotopy group} of $X$.
    \end{remark}
    \bigskip

    \begin{eg}
        \begin{itemize}
            \item Let $X$ be a convex subset of $\R^n$, then any two loops based at $x_0\in X$ are path homomorphic under the straight-line homotopy, implying that $\pi_1(X,x_0)$ is the trivial group containing a single element.
            \item Let $X=S^2$. Then $\pi_1(X,x_0)=\set{[f_k]:k\in\Z}\cong\Z$ where $f_0$ is the constant loop; $f_k$ represents $k$ clockwise turns for $k>0$; $f_k$ represents $-k$ counterclockwise turns for $k<0$. 
        \end{itemize}
    \end{eg}
    \bigskip

    \begin{defi}{$\hat{\alpha}$}
        Let $\alpha$ be a path in $X$ from $x_0$ to $x_1$. We define a map
        \[
            \hat{\alpha}:\pi_1(X,x_0)\to\pi_1(X,x_1)
        \]
        by
        \[
            \hat{\alpha}([f])=[\bar{\alpha}]*[f]*[\alpha]
        \]
    \end{defi}
    \bigskip

    \begin{thm}
        The map $\hat{\alpha}$ is a group isomorphism.
    \end{thm}
    \begin{cor}
        If $X$ is path connected, then
        \[
            \pi_1(X,x_0)\cong\pi_1(X,x_1)
        \]
        for any $x_0,x_1\in X$.
    \end{cor}
    \begin{remark}
        Let $x_0\in C\subset X$ where $C$ is a path component of $X$, then we have
        \[
            \pi_1(C,x_0)=\pi_1(X,x_0)
        \]
        So we usually deal with only path-connected spaces for fundamental groups.
    \end{remark}
    \bigskip

    \begin{defi}{Simple Connected}
        A space $X$ is said to be \textbf{simple connected} if it is path connected and $\pi_1(X,x_0)$ is the trivial group, denoted $\pi_1(X,x_0)=0$, for some, and hence for all $x_0\in X$.
    \end{defi}
    \begin{lemma}
        The space $X$ is simply connected if and only if any two paths having the same initial and final points are path homotopic.
    \end{lemma}
    \bigskip

    \begin{defi}{Induced homomorphism}
        We denote
        \[
            h:(X,x_0)\to (Y,y_0)
        \]
        if $h:X\to Y$ is continuous and $h(x_0)=y_0$. It induces a homomorphism
        \[
            h_*:\pi_1(X,x_0)\to\pi_1(Y,y_0)
        \]
        defined by
        \[
            h_*([f])=[h\circ f]
        \]
        often denoted as $(h_{x_0})_*$ to clearify the base point.
    \end{defi}
    \bigskip

    \begin{thm}[Functorial Properties]
        For continuous maps
        \[
            (X,x_0)\xrightarrow{h}(Y,y_0)\xrightarrow{k}(Z,z_0)
        \]
        we have
        \[
            (k\circ h)_*=k_*\circ h_*
        \]
        For the identity map \(i:(X,x_0)\to(X,x_0)\)
        \[
            i_*=\operatorname{id}_{\pi_1(X,x_0)}
        \]
    \end{thm}
    \begin{cor}
        If $h:(X,x_0)\to (Y,y_0)$ is a homemorphism of $X$ with $Y$, then $h_*$ is an isomorphism of $\pi_1(X,x_0)$ with $\pi_1(Y,y_0)$.
    \end{cor}
    \bigskip

    \begin{ex}
        A subset $A$ of $\R^n$ is said to be \textbf{star convex} if for some $a_0\in A$, all the line segments joining $a_0$ to other points of $A$ lie in $A$.
        \begin{itemize}
            \item Find a star convex set that is not convex.
            \begin{proof}
                The set $\set{(x,y):xy=0}\subset\R^2$ is star convex but not convex.
            \end{proof}
            \item Show that if $A$ is star convex, $A$ is simply connected.
            \begin{proof}
                For all $x\in A$. The line segement $\set{\alpha x+(1-\alpha)a_0:0\leq\alpha\leq1}$ is a path from $a_0$ to $x$ fully contained in $A$. Therefore, $A$ is path connected.

                Let $l$ be a loop based on $a_0$. Then $F$ defined by
                \[
                    F(s,t)=ta_0+(1-t)l(s)
                \]
                is a path homotopy from $l$ to the identitfy loop at $a_0$. It follows that $\pi_1(A,a_0)=0$.
            \end{proof}
        \end{itemize}
    \end{ex}
    \bigskip

    \begin{ex}
        Let $\alpha$ be a path in $X$ from $x_0$ to $x_1$; let $\beta$ be a path in $X$ from $x_1$ to $x_2$. Show that if $\gamma=\alpha*\beta$, then $\hat{\gamma}=\hat{\beta}\circ\hat{\alpha}$
    \end{ex}
    \begin{proof}
        Let $[f]\in \pi_1(X,x_0)$. Then
        \[
            \hat{\beta}(\hat{\alpha}([f]))=\hat{\beta}([\bar{\alpha}]*[f]*[\alpha])=[\bar{\beta}]*[\bar{\alpha}]*[f]*[\alpha]*[\beta]=[\bar{\gamma}]*[f]*[\gamma]=\hat{\gamma}([f])
        \]
    \end{proof}
    \bigskip

    \begin{ex}
        Let $X$ be path-connected and let $x_0,x_1\in X$. Show that $\pi_1(X,x_0)$ is abelian if and only if for every pair $\alpha$ and $\beta$ of paths from $x_0$ to $x_1$, we have $\hat{\alpha}=\hat{\beta}$.
    \end{ex}
    \begin{proof}
        $\impliedby:$ Let $[f],[g]\in\pi_1(X,x_0)$; let $\alpha$ be a path from $x_0$ to $x_1$ and $\beta=f*\alpha$. Then
        \[
            \hat{\beta}([g])=[\bar{\beta}]*[g]*[\beta]=[\bar{\alpha}]*[\bar{f}]*[g]*[f]*[\alpha]
        \]
        \[
            \hat{\alpha}([g])=[\bar{\alpha}]*[g]*[\alpha]
        \]
        It follows that
        \[
            [\bar{f}]*[g]*[f]=[g]\implies [g]*[f]=[f]*[g]
        \]

        $\implies:$ Let $\alpha,\beta$ be paths from $x_0$ to $x_1$. Let $[f]\in\pi_1(X,x_0)$. Then we have
        \[
            \hat{\alpha}([f])=[\bar{\alpha}]*[f]*[\alpha]=[\bar{\alpha}]*([\beta]*[\bar{\beta}])*[f]*([\beta]*[\bar{\beta}])*[\alpha]
        \]
        We know $\pi_1(X,x_1)$ is also abelian. Then
        \begin{align*}
            &([\bar{\alpha}]*[\beta]),([\bar{\beta}]*[f]*[\beta]),([\bar{\beta}]*[\alpha])\in\pi_1(X,x_1)\\
            \implies&\hat{\alpha}([f])=([\bar{\alpha}]*[\beta])*([\bar{\beta}]*[\alpha])*([\bar{\beta}]*[f]*[\beta])=[\bar{\beta}]*[f]*[\beta]=\hat{\beta}([f])
        \end{align*}
    \end{proof}
    \bigskip

    \begin{ex}
        Let $A\subset X$; let $r:X\to A$ be a retraction of $X$ onto $A$ (i.e. $r$ is continuous and $r(a)=a$ for all $a\in A$). If $a_0\in A$, show that
        \[
            r_*:\pi_1(X,a_0)\to \pi_1(A,a_0)
        \]
        is surjective.
    \end{ex}
    \begin{proof}
        Let $p:I\to A$ be a loop at $a_0$. Then $p':I\to X$ defined by $p'=p$ is also a loop at $a_0$. Hence $r_*([p'])=[r\circ p']=[p]$ implying that $r_*$ is surjective.
    \end{proof}
    \bigskip

    \begin{ex}
        Show that if $X$ is path-connected, the homomorphism induced by a continuous map is independent of the base point, up to isomorphisms of the groups involved. More precisely, let $h:X\to Y$ be continuous, with $h(x_0)=y_0$ and $h(x_1)=y_1$. Let $\alpha$ be a path in $X$ from $x_0$ to $x_1$, let $\beta=h\circ\alpha$. Show that 
        \[
            \hat{\beta}\circ(h_{x_0})_*=(h_{x_1})_*\circ\hat{\alpha}
        \]
    \end{ex}
    \begin{proof}
        Given $[f]\in\pi_1(X,x_0)$, we have
        \[
            \hat{\beta}\circ(h_{x_0})_*[f]=\hat{\beta}([h(f)])=[\bar{\beta}][h(f)][\beta]
        \]
        \[
            (h_{x_1})_*\circ\hat{\alpha}[f]=(h_{x_1})_*([\bar{\alpha}][f][\alpha])=[h(\bar{\alpha})][h(f)][h\alpha]=[\bar{\beta}][h(f)][\beta]
        \]
    \end{proof}
    \bigskip

    \begin{ex}
        Let $\group{G,\cdot}$ be a topological group with identity $x_0$. Let $\Omega(G,x_0)$ denote the set of all loops in $G$ based at $x_0$. If $f,g\in \Omega(G,x_0)$, then $f\otimes g$ defines a loop by
        \[
            (f\otimes g)(s)=f(s)\cdot g(s)
        \]
        \begin{itemize}
            \item Show that $\Omega(G,x_0)$ together with the operation $\otimes$ is a group.
            \begin{proof}
                For all $f,g\in\Omega(G,x_0)$, by definition we have $f\otimes g\in \Omega(G,x_0)$.

                For all $f,g,h\in\Omega(G,x_0)$, we have
                \[
                    (f\otimes g)\otimes h(s)=f\otimes(g\otimes h)(s)=f(s)g(s)h(s)
                \]

                Let $e_{x_0}\in \Omega(G,x_0)$ be the constant loop based at $x_0$. Then for $f\in\Omega(G,x_0)$, since $x_0$ is the identity, we have
                \[
                    (f\otimes e_{x_0})(s)=f(s)x_0=f(s)=(e_{x_0}\otimes f)(s)
                \]
                then $e_{x_0}$ is the identity of $\Omega(G,x_0)$.

                Let $f\in \Omega(G,x_0)$, then $f':I\to G$ defined by
                \[
                    f'(s)=f^{-1}(s)
                \]
                is trivially the inverse of $f$.
            \end{proof}
            \item Show that this operation induces group operation $\otimes$ on $\pi_1(G,x_0)$ which is identical to $*$. More precisely, let $f,g\in\Omega(G,x_0)$, then
            \[
                [f*g]=[f\otimes g]
            \]
            \begin{proof}
                To show that $[f]\otimes [g]=[f\otimes g]$ is well-defined. Let $F$ be the path homotopy between $f$ and $f'$, $G$ be the path homotopy between $g$ and $g'$. Then $H$ defined by
                \[
                    H(s,t)=F(s,t)\cdot G(s,t)
                \]
                can be verified to be the path homotopy between $f\otimes g$ and $f'\otimes g'$.

                Therefore, we have
                \[
                    [f\otimes g]=[f]\otimes[g]=[f*e_{x_0}]\otimes[e_{x_0}*g]=[(f*e_{x_0})\otimes(e_{x_0}*g)]=[f*g]
                \]
            \end{proof}
            \item Show that $\pi_1(G,x_0)$ is abelian.
            \begin{proof}
                Let $f,g\in \Omega(G, x_0)$. Define $g',f'$ by $g'(s)=g^{-1}(s)$ and $f'(s)=f^{-1}(s)$. Then
                \[
                    [\bar{f}][f]=[f'][f]=[e_{x_0}]\implies [\bar{f}]=[f']
                \]
                It follows that
                \[
                    [f]*[g]*[\bar{f}]*[\bar{g}]=[(f*g)\otimes(f'*g')]=[e_{x_0}]\implies [f][g]=[g][f]
                \]
            \end{proof}
        \end{itemize}
    \end{ex}
